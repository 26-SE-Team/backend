CREATE TABLE `user` (
    `user_id` BIGINT NOT NULL,
    `social_type` ENUM('GOOGLE', 'KAKAO') NULL,
    `google_email` VARCHAR(100) NULL,
    `kakao_email` VARCHAR(100) NULL,
    `name` VARCHAR(50) NOT NULL,
    `phone` VARCHAR(30) NULL,
    `role` ENUM('USER', 'ADMIN') NOT NULL DEFAULT 'USER',
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NULL,
    PRIMARY KEY (`user_id`)
);

CREATE TABLE `agent_profile` (
    `user_id` BIGINT NOT NULL,
    `license_no` VARCHAR(100) NOT NULL,
    `verification_status` ENUM('PENDING', 'APPROVED', 'REJECTED') NOT NULL DEFAULT 'PENDING',
    `rejection_reason` VARCHAR(255) NULL,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NULL,
    PRIMARY KEY (`user_id`)
);

CREATE TABLE `space` (
    `space_id` BIGINT NOT NULL,
    `agent_id` BIGINT NOT NULL,
    `title` VARCHAR(100) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    `area` DOUBLE NULL,
    `deposit` INT NULL,
    `monthly_rent` INT NULL,
    `maintenance_fee` INT NULL,
    `room_type` VARCHAR(50) NULL,
    `available_date` DATE NULL,
    `status` VARCHAR(20) NULL,
    `living_environment_info` TEXT NULL,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NULL,
    PRIMARY KEY (`space_id`)
);

CREATE TABLE `space_image` (
    `image_id` BIGINT NOT NULL,
    `space_id` BIGINT NOT NULL,
    `image_order` INT NOT NULL,
    `image_url` VARCHAR(500) NOT NULL,
    PRIMARY KEY (`image_id`)
);

CREATE TABLE `three_d_model` (
    `model_id` BIGINT NOT NULL,
    `space_id` BIGINT NOT NULL,
    `model_url` VARCHAR(500) NOT NULL,
    `status` VARCHAR(20) NOT NULL,
    `created_at` DATETIME NULL,
    PRIMARY KEY (`model_id`)
);

CREATE TABLE `chat_room` (
    `chat_room_id` BIGINT NOT NULL,
    `space_id` BIGINT NOT NULL,
    `tenant_id` BIGINT NOT NULL,
    `created_at` DATETIME NOT NULL,
    PRIMARY KEY (`chat_room_id`)
);

CREATE TABLE `chat_message` (
    `message_id` BIGINT NOT NULL,
    `chat_room_id` BIGINT NOT NULL,
    `sender_id` BIGINT NOT NULL,
    `message` TEXT NOT NULL,
    `sent_at` DATETIME NOT NULL,
    PRIMARY KEY (`message_id`)
);

CREATE TABLE `favorite` (
    `favorite_id` BIGINT NOT NULL,
    `user_id` BIGINT NOT NULL,
    `space_id` BIGINT NOT NULL,
    `created_at` DATETIME NOT NULL,
    PRIMARY KEY (`favorite_id`)
);

CREATE TABLE `furniture_placement` (
    `placement_id` BIGINT NOT NULL,
    `user_id` BIGINT NOT NULL,
    `space_id` BIGINT NOT NULL,
    `layout_data` JSON NULL,
    `preview_image_url` VARCHAR(500) NULL,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NULL,
    PRIMARY KEY (`placement_id`)
);

CREATE TABLE `report` (
    `report_id` BIGINT NOT NULL,
    `reporter_id` BIGINT NOT NULL,
    `space_id` BIGINT NULL,
    `target_user_id` BIGINT NULL,
    `reason` VARCHAR(255) NOT NULL,
    `description` TEXT NULL,
    `status` ENUM('PENDING', 'RESOLVED', 'REJECTED') NOT NULL DEFAULT 'PENDING',
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NULL,
    PRIMARY KEY (`report_id`)
);

CREATE TABLE `notification` (
    `notification_id` BIGINT NOT NULL,
    `user_id` BIGINT NOT NULL,
    `type` VARCHAR(50) NOT NULL,
    `title` VARCHAR(100) NOT NULL,
    `content` VARCHAR(255) NOT NULL,
    `is_read` BOOLEAN NOT NULL DEFAULT FALSE,
    `created_at` DATETIME NOT NULL,
    PRIMARY KEY (`notification_id`)
);
